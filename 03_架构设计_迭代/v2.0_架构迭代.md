### 2.1 索引策略

```python
# 三层索引
class KnowledgeIndex:
    def __init__(self):
        # 1. 关键词倒排索引（精确匹配）
        self.inverted_index = {}  # word → [doc_id]
        # 2. 主题索引（树状）
        self.subject_tree = {}  # subject → topic → [doc_id]
        # 3. 向量索引（语义）
        self.vector_index = ChromaClient()  # doc_id → vector
```

### 2.2 检索算法（混合）

```python
def hybrid_search(query, learner, top_k=5):
    # 1. 关键词检索（BM25）
    keyword_results = self.inverted_index.bm25_search(query, top_k=20)
    
    # 2. 主题过滤
    subject_results = self.subject_tree.search(learner.current_subject)
    
    # 3. 向量检索
    vector_results = self.vector_index.search(query, top_k=20)
    
    # 4. RRF（Reciprocal Rank Fusion）融合
    combined = reciprocal_rank_fusion([
        keyword_results,
        subject_results,
        vector_results
    ])
    
    # 5. 适配画像重排
    reranked = self.profile_rerank(combined, learner, top_k=top_k)
    
    return reranked
```

### 2.3 知识库版本控制

```python
# 每个节点都有版本
class VersionedNode:
    node_id: str
    versions: List[NodeVersion]  # 历史版本
    current: NodeVersion
    parents: List[str]  # 父节点（前置知识）
    children: List[str]  # 子节点（后续知识）
    
    def update(self, new_content, author, commit_msg):
        new_v = NodeVersion(
            content=new_content,
            author=author,
            commit_msg=commit_msg,
            parent=self.current
        )
        self.versions.append(new_v)
        self.current = new_v
        return new_v.version_id
```

### 3.1 知识库更新

```python
class KnowledgeUpdater:
    def add_concept(self, concept: SubjectNode, source: str, confidence: float):
        """添加新概念。"""
        # 1. 来源验证
        if not self.verify_source(source):
            raise ValueError(f"来源不可信: {source}")
        # 2. 写新版本
        version_id = self.knowledge_kb.add(concept, source=source, confidence=confidence)
        # 3. 触发 A/B 测试
        if confidence < 0.8:
            self.ab_test_queue.put(concept, version_id)
        # 4. 写 git commit
        self.versioned_store.commit(
            message=f"Add concept: {concept.id}",
            author="PAEG_KnowledgeUpdater"
        )
        return version_id
    
    def update_concept(self, concept_id, updates, source):
        """更新已有概念。"""
        # 1. 查当前版本
        current = self.knowledge_kb.get(concept_id)
        # 2. 合并更新
        new_content = merge(current, updates, source=source)
        # 3. 写新版本（不删除旧版）
        new_version = self.knowledge_kb.update(concept_id, new_content)
        # 4. 触发回滚策略
        self.schedule_review(new_version, after_n_uses=20)
        return new_version
```

```python
class ProfileUpdater:
    def update_progress(self, learner_id, subject, score):
        """更新掌握度。"""
        profile = self.profile_store.get(learner_id)
        # 指数移动平均
        alpha = 0.3
        profile.subjects[subject].mastery = (
            alpha * score + (1 - alpha) * profile.subjects[subject].mastery
        )
        # 更新进度
        self.profile_store.put(profile)
        # 同步云端（异步）
        self.cloud_sync.async_push(learner_id, profile)
    
    def update_emotion(self, learner_id, signal):
        """更新情绪状态。"""
        profile = self.profile_store.get(learner_id)
        profile.emotion_history.append({
            "signal": signal,
            "timestamp": now()
        })
        # 保持最近 100 条
        profile.emotion_history = profile.emotion_history[-100:]
        # 趋势判断
        if len(profile.emotion_history) >= 5:
            recent = profile.emotion_history[-5:]
            if all(e["signal"] in {"frustrated", "bored"} for e in recent):
                profile.emotion_trend = "declining"
                # 触发家长通知（如果已授权）
                if profile.privacy.parent_notify_enabled:
                    self.parent_notify.send(learner_id, "持续挫败，建议关注")
        self.profile_store.put(profile)
```

```python
        cloud_response = self.cloud.put_changes(profile.id, local_changes)
```

## 8. 自我更新的"失败回滚"

```python
class SafeUpdater:
    def update_with_rollback(self, update_func, target_object, test_period_days=7):
        """带回滚的更新。"""
        # 1. 备份当前版本
        backup = self.store.get(target_object)
        # 2. 应用更新
        new_version = update_func(target_object)
        # 3. 设置观察期
        self.observation_queue.add(new_version, test_period_days)
        # 4. 在观察期内监控关键指标
        # 5. 观察期结束：自动对比新 vs 旧
        #    - 如果新版本表现更好：保留
        #    - 如果新版本表现更差：自动回滚
        if not self.evaluate_new_version(new_version, backup):
            self.store.rollback(target_object, backup.version_id)
            self.metacog_log.append(f"Rollback: {new_version.version_id}")
```

```python
class PAEGStateMachine:
    states = ["IDLE", "ONBOARD", "DIAGNOSE", "PLAN", "TEACH_LOOP", "SUMMARIZE", "REFLECT", "END"]
    
    transitions = {
        "IDLE": ["ONBOARD", "CLASSIFY_INPUT"],
        "ONBOARD": ["DIAGNOSE", "CHITCHAT"],
        "DIAGNOSE": ["PLAN", "ONBOARD"],  # 诊断可能不充分回到 onboarding
        "PLAN": ["TEACH_LOOP"],
        "TEACH_LOOP": ["PRESENT", "PRACTICE", "EVALUATE", "ADAPT", "SUMMARIZE"],
        "PRESENT": ["PRACTICE", "EVALUATE", "ADAPT"],
        "PRACTICE": ["EVALUATE", "ADAPT"],
        "EVALUATE": ["ADAPT", "PRESENT", "SUMMARIZE"],
        "ADAPT": ["PRESENT", "PRACTICE", "SUMMARIZE", "END"],
        "SUMMARIZE": ["REFLECT"],
        "REFLECT": ["END"],
        "END": ["IDLE"]
    }
    
    def transition(self, current_state, event):
        if event in self.transitions[current_state]:
            return event
        else:
            raise InvalidTransition(current_state, event)
```
